## 🚀 Quick Start

### 1. Integrate into App.tsx

```tsx
import { SettingsProvider } from './contexts/SettingsContext'
import SettingsPanel from './components/settings/SettingsPanel.refactored'

function App() {
  return (
    <SettingsProvider>
      {/* Application content */}
      <SettingsPanel
        isOpen={showSettings}
        onClose={() => setShowSettings(false)}
      />
    </SettingsProvider>
  )
}
```

### 2. Use Settings in Components

```tsx
import { useSettings } from './hooks/useSettings'

function MyComponent() {
  const { settings, updateSetting } = useSettings()
  return (
    <div>
      <p>Current language: {settings.language.locale}</p>
      <button onClick={() => updateSetting('language', 'locale', 'en-US')}>
        Switch to English
      </button>
    </div>
  )
}
```

### 3. Apply Appearance Settings

```tsx
import { useAppearance } from './hooks/useAppearance'

function App() {
  const { theme, fontSize } = useAppearance()
  return (
    <div className={theme} style={{ fontSize }}>
      {/* Application content */}
    </div>
  )
}
```
